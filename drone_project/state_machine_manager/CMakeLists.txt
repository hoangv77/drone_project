cmake_minimum_required(VERSION 3.8)
project(state_machine_manager)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(std_srvs REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(px4_interface_node REQUIRED)
find_package(safety_monitor REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(rosidl_typesupport_cpp REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/MissionState.msg" "msg/StageFeedback.msg"
  DEPENDENCIES builtin_interfaces)
rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} "rosidl_typesupport_cpp")

add_library(mission_machine src/mission_machine.cpp)
target_compile_features(mission_machine PUBLIC cxx_std_17)
target_include_directories(mission_machine PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
ament_target_dependencies(mission_machine px4_interface_node safety_monitor)
target_link_libraries(mission_machine "${cpp_typesupport_target}")
add_executable(state_machine_manager_node src/state_machine_manager.cpp)
set_target_properties(state_machine_manager_node PROPERTIES OUTPUT_NAME state_machine_manager)
target_compile_features(state_machine_manager_node PUBLIC cxx_std_17)
ament_target_dependencies(state_machine_manager_node rclcpp rcl_interfaces std_srvs)
target_link_libraries(state_machine_manager_node mission_machine)
install(TARGETS mission_machine
  EXPORT export_mission_machine
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(TARGETS state_machine_manager_node RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_export_include_directories(include)
ament_export_libraries(mission_machine)
ament_export_targets(export_mission_machine HAS_LIBRARY_TARGET)
ament_export_dependencies(rosidl_default_runtime px4_interface_node safety_monitor)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_mission_machine test/test_mission_machine.cpp)
  target_link_libraries(test_mission_machine mission_machine)
  find_package(launch_testing_ament_cmake REQUIRED)
  add_launch_test(test/test_mission_node.py TIMEOUT 120)
endif()
ament_package()
